CREATE   PROCEDURE dbo.usp_insert_daily_sales
    @MinRows INT = 30,
    @MaxRows INT = 100
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @NumRows INT = @MinRows + CAST(RAND(CHECKSUM(NEWID())) * (@MaxRows - @MinRows) AS INT);
    DECLARE @TodayUTC DATE = CAST(SYSUTCDATETIME() AS DATE);
    DECLARE @ProdCount INT = (SELECT COUNT(*) FROM dbo.Product);
    DECLARE @StoreCount INT = (SELECT COUNT(*) FROM dbo.Store);
    DECLARE @CustCount INT = (SELECT COUNT(*) FROM dbo.Customer);

    ;WITH Numbers AS (
        SELECT TOP (@NumRows) ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS n
        FROM sys.all_objects
    ),
    ProductRanked AS (
        SELECT prod_num, ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS rn FROM dbo.Product
    ),
    StoreRanked AS (
        SELECT store_num, ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS rn FROM dbo.Store
    ),
    CustomerRanked AS (
        SELECT cust_num, ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS rn FROM dbo.Customer
    )
    INSERT INTO dbo.f_sls_t (prod_num, store_num, cust_num, trxn_dt, qty, amount)
    SELECT
        p.prod_num,
        s.store_num,
        c.cust_num,
        DATEADD(SECOND, ABS(CHECKSUM(NEWID(), n.n)) % 86400, CAST(@TodayUTC AS DATETIME2)),
        q.qty,
        CAST(q.qty * q.unit_price AS DECIMAL(10,2))
    FROM Numbers n
    CROSS APPLY (SELECT prod_num FROM ProductRanked WHERE rn = 1 + ABS(CHECKSUM(NEWID(), n.n, 1)) % @ProdCount) p
    CROSS APPLY (SELECT store_num FROM StoreRanked WHERE rn = 1 + ABS(CHECKSUM(NEWID(), n.n, 2)) % @StoreCount) s
    CROSS APPLY (SELECT cust_num FROM CustomerRanked WHERE rn = 1 + ABS(CHECKSUM(NEWID(), n.n, 3)) % @CustCount) c
    CROSS APPLY (
        SELECT
            CAST(1 + (RAND(CHECKSUM(NEWID(), n.n, 4)) * 4) AS INT) AS qty,
            CAST(5 + (RAND(CHECKSUM(NEWID(), n.n, 5)) * 95) AS DECIMAL(10,2)) AS unit_price
    ) q
    OPTION (MAXDOP 1);

    PRINT CONCAT('Inserted ', @NumRows, ' sales rows for ', CONVERT(VARCHAR(10), @TodayUTC, 120), ' (UTC)');
END
